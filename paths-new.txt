export const buildRoleEditPath = (
    name: string,
    userTypeRole: string,
    applicationRoleId: string
) =>
    paths.ROLE_EDIT.pathName
        .replace(":name", encodeURIComponent(name))
        .replace(":type", encodeURIComponent(userTypeRole))
        .replace(":id", encodeURIComponent(applicationRoleId));
